```ts
export async function extendTakeoverOnHumanMessage(args: {
  companyId: string; leadId: string; userId: string; extendHours?: number;
}): Promise<void> {
  const admin = createAdminClient();
  const until = new Date(Date.now() + (args.extendHours ?? DEFAULT_HOURS) * 3_600_000).toISOString();
  // WHERE guard: no-op if lead is not currently in an active takeover
  await admin.from('leads').update({
    human_takeover_until: until,
    human_takeover_by: args.userId,
  })
    .eq('id', args.leadId)
    .eq('company_id', args.companyId)
    .not('human_takeover_until', 'is', null)
    .gt('human_takeover_until', new Date().toISOString());
}
```

```ts
export async function setControlMode(leadId: string, mode: 'autonomous' | 'shadow' | 'manual') {
  const profile = await getUserProfile();
  if (!profile) throw new Error("Unauthorized");

  const supabase = await createClient();
  const { company_id } = await getLeadInfo(supabase, leadId);
  await requireOnboarding(company_id);

  if (mode === 'manual') {
    // Full takeover: sets is_paused + human_takeover_until (engine guard depends on this)
    await activateTakeover({ companyId: company_id, leadId, userId: profile.id, durationHours: 24 });
  } else {
    // Autonomous or shadow: clear takeover, set control_mode directly after
    await deactivateTakeover({ companyId: company_id, leadId });
    if (mode === 'shadow') {
      // deactivateTakeover sets 'autonomous'; override to 'shadow'
      await supabase.from('leads').update({ control_mode: 'shadow', is_paused: false }).eq('id', leadId);
    }
  }

  const modeLabels = {
    autonomous: "automático (Modo Autônomo)",
    shadow: "Copiloto / Shadow (rascunhos inteligentes)",
    manual: "manual (Modo Manual)"
  };

  const { error } = await supabase.from("messages").insert({
    lead_id: leadId,
    company_id,
    content: `Modo de controle alterado para ${modeLabels[mode]}.`,
    role: "note",
  });

  if (error) throw new Error(error.message);

  revalidatePath("/inbox");
}
```

In `ReportsPage`, after `const companyId = profile?.memberships?.[0]?.company_id;`, add the plan check and ai_logs fetch. Place this BEFORE the existing `Promise.all`:

```ts
  // Plan limits — determines whether to show provider health section
  const adminClient = createAdminClient();
  const { data: companyData } = await adminClient
    .from('companies')
    .select('plan_type')
    .eq('id', companyId)
    .single();
  const planLimits = getPlanLimits((companyData?.plan_type ?? 'trial') as PlanType);
  const hasAnalytics = planLimits.hasAnalytics ?? false;

  // Provider health — only fetch for plans with analytics access
  let providerStats: ProviderStat[] = [];
  let chainStats: ChainStat[] = [];

  if (hasAnalytics) {
    const since7d = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000).toISOString();
    const since24h = new Date(Date.now() - 24 * 60 * 60 * 1000).toISOString();

    const { data: aiLogs } = await adminClient
      .from('ai_logs')
      .select('provider, chain_kind, latency_ms, cost, error, created_at')
      .eq('company_id', companyId)
      .gte('created_at', since7d)
      .not('provider', 'is', null);

    const logs = aiLogs ?? [];

    // Aggregate by provider
    const providerMap = new Map<string, {
      req7d: number; req24h: number; successes: number;
      latencySum: number; latencyCount: number; costSum: number;
    }>();

    const PROVIDERS = ['cerebras', 'groq', 'sambanova', 'gemini'] as const;
    for (const p of PROVIDERS) {
      providerMap.set(p, { req7d: 0, req24h: 0, successes: 0, latencySum: 0, latencyCount: 0, costSum: 0 });
    }

    for (const log of logs) {
      if (!log.provider) continue;
      const agg = providerMap.get(log.provider);
      if (!agg) continue;
      agg.req7d++;
      if (log.created_at >= since24h) agg.req24h++;
      if (!log.error) agg.successes++;
      if (log.latency_ms != null) { agg.latencySum += log.latency_ms; agg.latencyCount++; }
      if (log.cost != null) agg.costSum += log.cost;
    }

    providerStats = PROVIDERS.map((p) => {
      const agg = providerMap.get(p)!;
      return {
        provider: p,
        requests_24h: agg.req24h,
        requests_7d: agg.req7d,
        successes_7d: agg.successes,
        avg_latency_ms: agg.latencyCount > 0 ? Math.round(agg.latencySum / agg.latencyCount) : 0,
        total_cost_7d: parseFloat(agg.costSum.toFixed(6)),
      };
    });

    // Chain kind aggregation
    const chainMap = new Map<string, number>([['conv', 0], ['tools', 0], ['bg', 0]]);
    for (const log of logs) {
      if (log.chain_kind && chainMap.has(log.chain_kind)) {
        chainMap.set(log.chain_kind, (chainMap.get(log.chain_kind) ?? 0) + 1);
      }
    }
    chainStats = (['conv', 'tools', 'bg'] as const).map((k) => ({
      chain_kind: k,
      count: chainMap.get(k) ?? 0,
    }));
  }
```

```tsx
const PROVIDER_META: Record<string, { label: string; color: string; icon: string }> = {
  cerebras:  { label: 'Cerebras',   color: '#6366F1', icon: '⚡' },
  groq:      { label: 'Groq',       color: '#F59E0B', icon: '🔥' },
  sambanova: { label: 'SambaNova',  color: '#10B981', icon: '🧠' },
  gemini:    { label: 'Gemini',     color: '#3B82F6', icon: '✦' },
};

function successBadge(rate: number) {
  if (rate >= 0.95) return { label: `${Math.round(rate * 100)}%`, cls: 'bg-emerald-500/20 text-emerald-400' };
  if (rate >= 0.80) return { label: `${Math.round(rate * 100)}%`, cls: 'bg-yellow-500/20 text-yellow-400' };
  return { label: `${Math.round(rate * 100)}%`, cls: 'bg-red-500/20 text-red-400' };
}

function ProviderHealthSection({
  providerStats,
  chainStats,
}: {
  providerStats: ProviderStat[];
  chainStats: ChainStat[];
}) {
  const totalChain = chainStats.reduce((s, c) => s + c.count, 0);

  const chainColors: Record<string, string> = {
    conv: '#3B82F6',
    tools: '#8B5CF6',
    bg: '#6B7280',
  };
  const chainLabels: Record<string, string> = {
    conv: 'Conversação',
    tools: 'Ferramentas',
    bg: 'Background',
  };

  return (
    <section className="mt-8">
      <div className="mb-4 flex items-center gap-2">
        <Activity className="h-5 w-5 text-indigo-400" />
        <h2 className="text-lg font-semibold text-white">Saúde dos Providers IA</h2>
        <span className="ml-auto text-xs text-white/40">últimos 7 dias</span>
      </div>

      {/* Provider cards */}
      <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
        {providerStats.map((stat, i) => {
          const meta = PROVIDER_META[stat.provider];
          const rate = stat.requests_7d > 0 ? stat.successes_7d / stat.requests_7d : 1;
          const badge = successBadge(rate);
          const isEmpty = stat.requests_7d === 0;

          return (
            <motion.div
              key={stat.provider}
              initial={{ opacity: 0, y: 16 }}
              animate={{ opacity: 1, y: 0 }}
              transition={{ delay: i * 0.06, duration: 0.35 }}
              className="glass rounded-2xl p-4"
            >
              <div className="mb-3 flex items-center justify-between">
                <span className="text-lg">{meta.icon}</span>
                <span
                  className={cn(
                    'rounded-full px-2 py-0.5 text-xs font-semibold',
                    isEmpty ? 'bg-white/10 text-white/40' : badge.cls
                  )}
                >
                  {isEmpty ? 'Não usado' : badge.label}
                </span>
              </div>
              <p className="text-sm font-semibold text-white">{meta.label}</p>
              {isEmpty ? (
                <p className="mt-2 text-xs text-white/30">Sem chamadas nos últimos 7 dias</p>
              ) : (
                <div className="mt-2 space-y-1 text-xs text-white/60">
                  <div className="flex justify-between">
                    <span>Req 24h</span>
                    <span className="font-mono text-white">{stat.requests_24h}</span>
                  </div>
                  <div className="flex justify-between">
                    <span>Req 7d</span>
                    <span className="font-mono text-white">{stat.requests_7d}</span>
                  </div>
                  <div className="flex justify-between">
                    <span>Lat. média</span>
                    <span className="font-mono text-white">{stat.avg_latency_ms}ms</span>
                  </div>
                  <div className="flex justify-between">
                    <span>Custo 7d</span>
                    <span className="font-mono text-white">
                      {stat.total_cost_7d < 0.001
                        ? '<$0.001'
                        : `$${stat.total_cost_7d.toFixed(4)}`}
                    </span>
                  </div>
                </div>
              )}
            </motion.div>
          );
        })}
      </div>

      {/* Chain kind bar */}
      {totalChain > 0 && (
        <div className="glass mt-4 rounded-2xl p-4">
          <p className="mb-3 text-sm font-semibold text-white/80">Roteamento de Chain (7d)</p>
          <div className="flex h-3 w-full overflow-hidden rounded-full">
            {chainStats
              .filter((c) => c.count > 0)
              .map((c) => (
                <motion.div
                  key={c.chain_kind}
                  initial={{ width: 0 }}
                  animate={{ width: `${(c.count / totalChain) * 100}%` }}
                  transition={{ duration: 0.6, ease: 'easeOut' }}
                  style={{ backgroundColor: chainColors[c.chain_kind] }}
                  className="h-full"
                />
              ))}
          </div>
          <div className="mt-2 flex gap-4">
            {chainStats.map((c) => (
              <div key={c.chain_kind} className="flex items-center gap-1.5 text-xs text-white/60">
                <span
                  className="h-2 w-2 rounded-full"
                  style={{ backgroundColor: chainColors[c.chain_kind] }}
                />
                {chainLabels[c.chain_kind]}{' '}
                <span className="font-mono text-white">
                  {totalChain > 0 ? Math.round((c.count / totalChain) * 100) : 0}%
                </span>
              </div>
            ))}
          </div>
        </div>
      )}
    </section>
  );
}
```
